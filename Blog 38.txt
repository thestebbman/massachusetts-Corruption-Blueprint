from collections import Counter
import re

def generate_tags():
    stopwords = {"the", "and", "to", "of", "in", "a", "for", "on", "is", "with", "as", "by", "that"}
    tags_dict = {}
    
    for filename, content in memory_dict.items():
        words = re.findall(r'\b\w{4,}\b', content.lower())
        words = [w for w in words if w not in stopwords]
        common_words = Counter(words).most_common(5)
        tags_dict[filename] = [word for word, _ in common_words]
    
    with open("blog_tags.json", "w") as file:
        json.dump(tags_dict, file)
    
    print("Tags generated and saved to blog_tags.json")
