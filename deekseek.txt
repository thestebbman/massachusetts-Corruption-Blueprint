import sqlite3
from datetime import datetime
import os

class AIMemory:
    def __init__(self, db_path="ai_memory.db"):
        self.db_path = os.path.join('/data/ai_memory', db_path)
        os.makedirs(os.path.dirname(self.db_path), exist_ok=True)
        self.conn = sqlite3.connect(self.db_path)
        self._init_db()
        
    def _init_db(self):
        cursor = self.conn.cursor()
        cursor.execute('''
        CREATE TABLE IF NOT EXISTS memories (
            id INTEGER PRIMARY KEY,
            timestamp TEXT NOT NULL,
            category TEXT NOT NULL,
            content TEXT NOT NULL,
            importance INTEGER DEFAULT 1
        )
        ''')
        self.conn.commit()
    
    def add_memory(self, category, content, importance=1):
        cursor = self.conn.cursor()
        cursor.execute('''
        INSERT INTO memories (timestamp, category, content, importance)
        VALUES (?, ?, ?, ?)
        ''', (datetime.now().isoformat(), category, content, importance))
        self.conn.commit()
    
    def recall_memory(self, category=None, search_term=None):
        cursor = self.conn.cursor()
        query = "SELECT * FROM memories"
        params = []
        
        if category or search_term:
            query += " WHERE "
            conditions = []
            if category:
                conditions.append("category = ?")
                params.append(category)
            if search_term:
                conditions.append("content LIKE ?")
                params.append(f"%{search_term}%")
            query += " AND ".join(conditions)
            
        query += " ORDER BY timestamp DESC"
        cursor.execute(query, params)
        return cursor.fetchall()

#!/data/ai_venv/bin/python3
import sqlite3
from datetime import datetime
import os

class AIMemory:
    def __init__(self, db_path="ai_memory.db"):
        self.db_path = os.path.join('/data/ai_memory', db_path)
        os.makedirs(os.path.dirname(self.db_path), exist_ok=True)
        self.conn = sqlite3.connect(self.db_path)
        self._init_db()
        
    def _init_db(self):
        cursor = self.conn.cursor()
        cursor.execute('''
        CREATE TABLE IF NOT EXISTS memories (
            id INTEGER PRIMARY KEY,
            timestamp TEXT NOT NULL,
            category TEXT NOT NULL,
            content TEXT NOT NULL,
            importance INTEGER DEFAULT 1
        )
        ''')
        self.conn.commit()
    
    def add_memory(self, category, content, importance=1):
        cursor = self.conn.cursor()
        cursor.execute('''
        INSERT INTO memories (timestamp, category, content, importance)
        VALUES (?, ?, ?, ?)
        ''', (datetime.now().isoformat(), category, content, importance))
        self.conn.commit()
    
    def recall_memory(self, category=None, search_term=None):
        cursor = self.conn.cursor()
        query = "SELECT * FROM memories"
        params = []
        
        if category or search_term:
            query += " WHERE "
            conditions = []
            if category:
                conditions.append("category = ?")
                params.append(category)
            if search_term:
                conditions.append("content LIKE ?")
                params.append(f"%{search_term}%")
            query += " AND ".join(conditions)
            
        query += " ORDER BY timestamp DESC"
        cursor.execute(query, params)
        return cursor.fetchall()
