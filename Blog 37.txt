# ai_memory_assistant.py
# Built for Ricky: one brick, one brain, full control.

import chromadb
from chromadb.config import Settings
from datetime import datetime
import os

# Set up persistent ChromaDB
persist_dir = "./ai_memory"
os.makedirs(persist_dir, exist_ok=True)
client = chromadb.Client(Settings(chroma_db_impl="duckdb+parquet", persist_directory=persist_dir))

# Collection setup
if "memory" not in [c.name for c in client.list_collections()]:
    collection = client.create_collection(name="memory")
else:
    collection = client.get_collection(name="memory")

# Helper functions
def add_memory(text, tag=None):
    id = f"mem-{datetime.now().isoformat()}"
    metadata = {"tag": tag or "general", "timestamp": datetime.now().isoformat()}
    collection.add(documents=[text], ids=[id], metadatas=[metadata])
    print(f"✅ Memory added: {text}")

def search_memory(query):
    results = collection.query(query_texts=[query], n_results=5)
    print("\n🔍 Memory Search Results:")
    for doc, meta in zip(results['documents'][0], results['metadatas'][0]):
        print(f"- {doc} (tag: {meta['tag']}, time: {meta['timestamp']})")

def show_all_memories():
    print("\n📚 All Memories Stored:")
    all = collection.peek()
    for item in all:
        print(f"- {item['document']} (id: {item['id']})")

# Checklist support
def add_checklist(item):
    add_memory(item, tag="checklist")

def list_checklist():
    print("\n📝 Checklist Items:")
    results = collection.query(query_texts=["checklist"], n_results=100)
    for doc, meta in zip(results['documents'][0], results['metadatas'][0]):
        if meta['tag'] == "checklist":
            print(f"- {doc}")

def mark_done(item):
    add_memory(f"✔️ DONE: {item}", tag="checklist")

# Main CLI loop
def menu():
    while True:
        print("\n🧠 AI Memory Assistant")
        print("[1] Add Memory")
        print("[2] Search Memory")
        print("[3] Show All Memories")
        print("[4] Add Checklist Item")
        print("[5] Show Checklist")
        print("[6] Mark Checklist Item Done")
        print("[7] Exit")

        choice = input("Select: ")
        if choice == "1":
            add_memory(input("Enter memory: "))
        elif choice == "2":
            search_memory(input("Search query: "))
        elif choice == "3":
            show_all_memories()
        elif choice == "4":
            add_checklist(input("Enter checklist item: "))
        elif choice == "5":
            list_checklist()
        elif choice == "6":
            mark_done(input("Item completed: "))
        elif choice == "7":
            print("Goodbye.")
            break
        else:
            print("Invalid option.")

if __name__ == "__main__":
    menu()

from langchain.document_loaders import TextLoader
from langchain.chains import RetrievalQA
from langchain.llms import Ollama
import os

# Directory containing blog files
BLOG_DIR = "/mnt/data/blog_files"
# Persist directory for ChromaDB
PERSIST_DIR = "./ai_memory"
# Embedding model
EMBED_MODEL = "all-MiniLM-L6-v2"

# Step 1: Load blog files
def load_documents(directory):
    documents = []
    for filename in os.listdir(directory):
        if filename.endswith(".txt"):
            loader = TextLoader(os.path.join(directory, filename))
            documents.extend(loader.load())
    return documents

# Step 2: Split text into manageable chunks
def split_documents(documents):
    splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=50)
    return splitter.split_documents(documents)

# Step 3: Embed and persist documents using ChromaDB
def embed_and_persist(documents):
    vectorstore = Chroma.from_documents(
        documents,
        embedding=OpenAIEmbeddings(model=EMBED_MODEL),
        persist_directory=PERSIST_DIR
    )
    vectorstore.persist()
    return vectorstore

# Step 4: Load vectorstore for retrieval
def load_vectorstore():
    return Chroma(persist_directory=PERSIST_DIR, embedding_function=OpenAIEmbeddings(model=EMBED_MODEL))

# Step 5: Build the true AI partner QA chain
def build_qa_chain():
    retriever = load_vectorstore().as_retriever()
    llm = Ollama(model="gemma:3b")  # Replace with your preferred model
    qa_chain = RetrievalQA.from_chain_type(llm=llm, retriever=retriever)
    return qa_chain

from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain_community.vectorstores import Chroma
from langchain_community.document_loaders import TextLoader
from langchain_community.llms import Ollama
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.chains import RetrievalQA
import os

# Adjust these paths to match your setup
BLOG_DIR = "blog_files"
PERSIST_DIR = "ai_memory"
EMBED_MODEL = "sentence-transformers/all-MiniLM-L6-v2"

def load_documents(directory):
    documents = []
    for filename in os.listdir(directory):
        if filename.endswith(".txt"):
            path = os.path.join(directory, filename)
            loader = TextLoader(path, encoding="utf-8")
            documents.extend(loader.load())
    return documents

def split_documents(documents):
    splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=50)
    return splitter.split_documents(documents)

def embed_and_persist(documents):
    embeddings = HuggingFaceEmbeddings(model_name=EMBED_MODEL)
    vectorstore = Chroma.from_documents(
        documents,
        embedding=embeddings,
        persist_directory=PERSIST_DIR
    )
    vectorstore.persist()
    return vectorstore

def load_vectorstore():
    embeddings = HuggingFaceEmbeddings(model_name=EMBED_MODEL)
    return Chroma(persist_directory=PERSIST_DIR, embedding_function=embeddings)

def build_qa_chain():
    retriever = load_vectorstore().as_retriever()
    llm = Ollama(model="gemma:2b")  # Change to "gemma:7b" or any other model you have
    return RetrievalQA.from_chain_type(llm=llm, retriever=retriever)

# Execution
if __name__ == "__main__":
    print("Loading documents...")
    docs = load_documents(BLOG_DIR)
    print("Splitting documents...")
    split_docs = split_documents(docs)
    print("Embedding and saving memory...")
    embed_and_persist(split_docs)
    print("Memory saved. Starting question interface.")

    qa = build_qa_chain()

    while True:
        query = input("\nAsk your partner: ")
        if query.lower() in ["quit", "exit"]:
            break
        answer = qa.run(query)
        print("\n🤖:", answer)
